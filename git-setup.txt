# Git Setup Commands for Paradise Nursery
# Copy and paste these commands one by one (replace YOUR-USERNAME)

# Step 1: Initialize Git repository
git init

# Step 2: Add all files
git add .

# Step 3: Create initial commit
git commit -m "Initial commit: Paradise Nursery Shopping Application with React and Redux"

# Step 4: Rename branch to main
git branch -M main

# Step 5: Add remote repository (REPLACE YOUR-USERNAME WITH YOUR GITHUB USERNAME)
git remote add origin https://github.com/YOUR-USERNAME/paradise-nursery.git

# Step 6: Push to GitHub
git push -u origin main

# ============================================
# After pushing, enable GitHub Pages:
# ============================================
# 1. Go to: https://github.com/YOUR-USERNAME/paradise-nursery/settings/pages
# 2. Wait for GitHub Actions workflow to complete (check Actions tab)
# 3. Your site will be live at: https://YOUR-USERNAME.github.io/paradise-nursery/
